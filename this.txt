local f28

local function f29(p)
    local ch = f28(p)
    if not ch then return nil end
    return ch:FindFirstChild(v145.LockPart) or ch:FindFirstChild("HumanoidRootPart")
end

function f28(p)
    if not p then return nil end
    if typeof(p) == "Instance" and p:IsA("Player") then return p.Character end
    if typeof(p) == "Instance" and p:IsA("Model")  then return p end
    return nil
end

local function f30(p)
    local ch = f28(p); if not ch then return nil end
    return ch:FindFirstChild("HumanoidRootPart") or ch:FindFirstChild("Torso") or ch:FindFirstChild("UpperTorso")
end

local function f31(p)
    local ch = f28(p)
    local hum = ch and ch:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function f32(p)
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    local list = {}
    if localPlayer.Character then table.insert(list, localPlayer.Character) end
    local ch = f28(p); if ch then table.insert(list, ch) end
    rp.FilterDescendantsInstances = list
    return rp
end

local function f33(p)
    local lp, tp = f30(localPlayer), f30(p)
    if not lp or not tp then return math.huge end
    return (lp.Position - tp.Position).Magnitude
end

local function f34(part, params)
    if not v145.VisibleCheck then return true end
    local origin = currentCamera.CFrame.Position
    local dir = part.Position - origin
    if not params then
        params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local list = {}
        if localPlayer.Character then table.insert(list, localPlayer.Character) end
        for _, plr in ipairs(players:GetPlayers()) do
            if plr.Character and plr.Character:IsAncestorOf(part) then
                table.insert(list, plr.Character); break
            end
        end
        local model = part:FindFirstAncestorOfClass("Model")
        if model and model ~= localPlayer.Character then
            local has = false
            for _, v in ipairs(list) do
                if v == model then has = true; break end
            end
            if not has then table.insert(list, model) end
        end
        params.FilterDescendantsInstances = list
    end
    return workspace:Raycast(origin, dir, params) == nil
end

local function f35(part)
    local vp = currentCamera.ViewportSize
    local cx, cy = vp.X / 2, vp.Y / 2
    local sp, onScreen = currentCamera:WorldToViewportPoint(part.Position)
    if not onScreen then return false, math.huge end
    local d = math.sqrt((sp.X - cx) ^ 2 + (sp.Y - cy) ^ 2)
    return d <= v145.FOVRadius, d
end

local function f36(p)
    if not p or not f31(p) then return false end
    local tgt = f29(p)
    if not tgt then return false end
    if f33(p) > v145.MaxDistance then return false end
    local inFov = f35(tgt); if not inFov then return false end
    if v145.VisibleCheck and not f34(tgt) then return false end
    return true
end

local function f37()
    local best, bestD = nil, math.huge
    for _, plr in ipairs(players:GetPlayers()) do
        if plr ~= localPlayer and f31(plr) then
            local t = f29(plr)
            if t and f33(plr) <= v145.MaxDistance then
                local inFov, d = f35(t)
                if inFov and f34(t) and d < bestD then
                    best, bestD = plr, d
                end
            end
        end
    end
    for model in pairs(v149) do
        if model and model.Parent and f31(model) then
            local t = f29(model)
            if t and f33(model) <= v145.MaxDistance then
                local inFov, d = f35(t)
                if inFov and f34(t) and d < bestD then
                    best, bestD = model, d
                end
            end
        end
    end
    return best
end

local function f40(size, col)

local function f41(vp)

local function f42(hp)

-- ESP render
local function renderESP(data, target, hum, name, isNPC, cam, lpPos, maxDistSq, from)
    local pos = target.Position
    local distM = 0
    if lpPos then
        local diff = lpPos - pos
        local magSq = diff.X^2 + diff.Y^2 + diff.Z^2
        if magSq > maxDistSq then hideData(data) return end
        distM = floor(magSq ^ 0.5)
    end
    local sp, onScreen = cam:WorldToViewportPoint(pos)
    if not onScreen then hideData(data) return end
    data._vis = true

    local x = sp.X
    local hpRatio = clamp(hum.Health / max3(hum.MaxHealth, 1), 0, 1)
    local topY = cam:WorldToViewportPoint(pos + Vector3.new(0, 3.5, 0)).Y
    local botY = cam:WorldToViewportPoint(pos - Vector3.new(0, 3.2, 0)).Y
    local baseH = abs(botY - topY)
    local boxH  = baseH * v181.BoxScale
    local boxW  = boxH * 0.58
    local boxX  = x - boxW * 0.5
    local boxY  = topY - (boxH - baseH) * 0.5

    if v181.Box then
        data.Box.Position  = Vector2.new(boxX, boxY)
        data.Box.Size      = Vector2.new(boxW, boxH)
        data.Box.Color     = isNPC and color4 or v181.BoxColor
        data.Box.Thickness = v181.BoxThickness
        data.Box.Visible   = true
    else data.Box.Visible = false end

    if v181.HealthBar then
        local w  = v181.HealthBarWidth
        local hx = boxX - w - v181.HealthBarPad
        local fh = max3(1, boxH * hpRatio)
        local cl = f42(hpRatio)
        data.HBOut.Position = Vector2.new(hx - 1, boxY - 1)
        data.HBOut.Size     = Vector2.new(w + 2, boxH + 2)
        data.HBOut.Color    = color6
        data.HBOut.Visible  = true
        data.HBBg.Position  = Vector2.new(hx, boxY)
        data.HBBg.Size      = Vector2.new(w, boxH)
        data.HBBg.Visible   = true
        data.HBFill.Position = Vector2.new(hx, boxY + (boxH - fh))
        data.HBFill.Size     = Vector2.new(w, fh)
        if cl ~= data._hpClr then data.HBFill.Color = cl; data._hpClr = cl end
        data.HBFill.Visible = true
    else
        data.HBOut.Visible, data.HBBg.Visible, data.HBFill.Visible = false, false, false
    end

    if v181.Name then
        data.Label.Text = name
        data.Label.Color = isNPC and color4 or v181.NameColor
        data.Label.Position = Vector2.new(x, boxY - 18)
        data.Label.Visible = true
    else data.Label.Visible = false end

    if v181.Distance then
        if distM ~= data._distM then
            data._distM = distM
            data._distStr = format("[%d m]", distM)
        end
        data.Dist.Text = data._distStr
        data.Dist.Position = Vector2.new(x, boxY + boxH + 3)
        data.Dist.Visible = true
    else data.Dist.Visible = false end

    if v181.Line then
        local lc = isNPC and color4 or v181.LineColor
        local m = string.lower(v181.LineMode)
        local toPos
        if m == "top" then toPos = Vector2.new(x, boxY)
        elseif m == "center" then toPos = Vector2.new(x, boxY + boxH * 0.5)
        else toPos = Vector2.new(x, boxY + boxH) end
        data.Line.From = from; data.Line.To = toPos
        data.Line.Color = lc; data.Line.Thickness = v181.LineThickness
        data.Line.Visible = true
        data.HeadDot.Position = toPos; data.HeadDot.Color = lc
        data.HeadDot.Visible = true
    else
        data.Line.Visible, data.HeadDot.Visible = false, false
    end
end

local function f57()
    if not v181.Enabled then f50() return end
    local cam = workspaceService.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    local ch = localPlayer.Character
    local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
    local lpPos = hrp and hrp.Position
    local maxSq = v181.MaxDistance * v181.MaxDistance
    local fromVec = f41(vp)

    for plr, d in pairs(v148) do
        if not players:FindFirstChild(plr.Name) then
            removePlayer(plr)
        else
            local c  = plr.Character
            local ch2 = c and c:FindFirstChild("HumanoidRootPart")
            local hm  = c and c:FindFirstChildOfClass("Humanoid")
            if not ch2 or not hm or hm.Health <= 0 then hideData(d)
            else renderESP(d, ch2, hm, plr.DisplayName, false, cam, lpPos, maxSq, fromVec) end
        end
    end
    for model, d in pairs(v149) do
        if not model or not model.Parent then removeModel(model)
        elseif isPlayerName(model.Name) then removeModel(model)
        else
            local hrp2 = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
            local hm2  = model:FindFirstChildOfClass("Humanoid")
            if not hrp2 or not hm2 or hm2.Health <= 0 then hideData(d)
            else renderESP(d, hrp2, hm2, model.Name, true, cam, lpPos, maxSq, fromVec) end
        end
    end
end

local v127 = {}

local function register(key, comp, initialVal, stateFn)
    local o = { Value = initialVal, _comp = comp }
    function o:Set(val)
        self.Value = val
        if stateFn then stateFn(val) end
        if comp and comp.Set then
            pcall(function() comp:Set(val) end)
        end
    end
    v127[key] = o
    return o
end

local function serialize(t)
    local keys = {}
    for k in pairs(t) do table.insert(keys, k) end
    table.sort(keys)
    local out = { "{" }
    for i, k in ipairs(keys) do
        local v = t[k]
        local line
        if type(v) == "table" then
            line = string.format('    "%s": { "R": %d, "G": %d, "B": %d }', k, v.R or 0, v.G or 0, v.B or 0)
        elseif type(v) == "boolean" or type(v) == "number" then
            line = string.format('    "%s": %s', k, tostring(v))
        else
            line = string.format('    "%s": %q', k, tostring(v))
        end
        if i < #keys then line = line .. "," end
        table.insert(out, line)
    end
    table.insert(out, "}")
    return table.concat(out, "\n")
end

local function applyConfig(t)
    suppressNotify = true
    for k, v in pairs(t) do
        local e = v127[k]
        if e then
            local nv = v
            if k:find("Color") and type(v) == "table" then
                nv = Color3.fromRGB(v.R or 255, v.G or 255, v.B or 255)
            elseif k == "Menu_Keybind" and type(v) == "string" then
                nv = Enum.KeyCode[v]
            end
            if nv ~= nil then e:Set(nv) end
        end
    end
    suppressNotify = false
end

local function saveConfig()
    local out = {}
    for k, e in pairs(v127) do
        if e.Value ~= nil then
            if k:find("Color") then
                local c = e.Value
                out[k] = { R = math.floor(c.R * 255 + 0.5), G = math.floor(c.G * 255 + 0.5), B = math.floor(c.B * 255 + 0.5) }
            elseif k == "Menu_Keybind" then
                out[k] = e.Value.Name
            else
                out[k] = e.Value
            end
        end
    end
    local ok, err = pcall(function() writefile(CONFIG_FILE, serialize(out)) end)
    if ok then
        Notify({ Title = "Config Saved", Content = "Saved to " .. CONFIG_FILE, Duration = 2 })
    else
        Notify({ Title = "Save Failed", Content = tostring(err), Duration = 3 })
    end
end

local function resetConfig()
    applyConfig(v128)
    pcall(function()
        if delfile and isfile and isfile(CONFIG_FILE) then delfile(CONFIG_FILE) end
    end)
    Notify({ Title = "Config Reset", Content = "All settings restored to default", Duration = 2 })
end

local function loadConfig()
    local ok, content = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not content then applyConfig(v128); return end
    local ok2, decoded = pcall(function() return httpService:JSONDecode(content) end)
    if not ok2 or type(decoded) ~= "table" then applyConfig(v128); return end
    applyConfig(decoded)
end

-- ============================================================
-- Render loop
-- ============================================================
local smoothFactors = { 0.3, 0.15, 0.06 }

runService.RenderStepped:Connect(function()
    currentCamera = workspaceService.CurrentCamera
    if not currentCamera then return end

    -- FOV circle
    if v145.FOVEnabled then
        circle.Position = Vector2.new(currentCamera.ViewportSize.X / 2, currentCamera.ViewportSize.Y / 2)
        circle.Radius   = v145.FOVRadius
        circle.Color    = v146 and v145.FOVActiveColor or v145.FOVColor
        circle.Visible  = true
    else
        circle.Visible = false
    end

    -- ESP
    f57()

    -- Aimbot
    if not v145.LockEnabled then
        if v146 then Unlock() end
        return
    end
    if not v146 then
        local t = f37()
        if t then f38(t) end
        return
    end
    if not f31(v174) then Unlock() return end

    local tgt = f29(v174)
    if not tgt then Unlock() return end
    if f33(v174) > v145.MaxDistance then Unlock() return end

    if v176 ~= v174 then
        v176 = v174
        v175 = f32(v174)
    end

    v147 = v147 + 1
    if v147 >= 3 then
        v147 = 0
        if not f34(tgt, v175) then
            Unlock()
            local newT = f37()
            if newT then f38(newT) else circle.Color = v145.FOVColor end
            return
        end
    end

    local camPos  = currentCamera.CFrame.Position
    local look    = currentCamera.CFrame.LookVector
    local targetD = (tgt.CFrame.Position - camPos).Unit
    local newLook = look:Lerp(targetD, smoothFactors[v145.SmoothLevel] or 0.15)
    currentCamera.CFrame = CFrame.new(camPos, camPos + newLook)
end)
