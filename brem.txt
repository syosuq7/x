S.scanRadius = function()
    local base = S.OwnershipRadius or 200
    if S.FarGrabEnabled then local far = S.FarGrabRadius or 450; if far > base then return far end end
    return base
end
S.lostReturnDist = function()
    local d = S.ReturnDistance or 260
    if S.FarGrabEnabled then local w = (S.scanRadius() or 0) + 100; if w > d then d = w end end
    return d
end
S.farDrag = function(part, cp, tp, ds)
    local dir = tp - cp
    if dir.Magnitude > 0.05 then dir = dir.Unit else dir = Vector3.zero end
    local fs = math.min(S.FarGrabTopSpeed or 260, ds * 2.2)
    if fs < 60 then fs = 60 end
    part.CanCollide = false
    part.AssemblyLinearVelocity = dir * fs
end
S.boostSimRadius = function(override)
    local base = math.max(S.OwnershipRadius or 200, 10)
    local want = override or (base * math.clamp(S.ClaimBoost or 1.6, 1, 5))
    if (not override) and S.escalatedRadius then want = math.max(want, S.escalatedRadius) end
    local cap = S.SimRadMax or 4000
    if not override then
        if S.SimRadMin and want < S.SimRadMin then want = S.SimRadMin end
        if S.scanRadius and want < S.scanRadius() then want = S.scanRadius() end
    end
    if want > cap then want = cap end
    if want < 10 then want = 10 end
    S.currentBoostRadius = want
    pcall(function() LocalPlayer.MaximumSimulationRadius = cap end)
    pcall(function() sethiddenproperty(LocalPlayer, "MaximumSimulationRadius", cap) end)
    pcall(function() LocalPlayer.SimulationRadius = want end)
    pcall(function() sethiddenproperty(LocalPlayer, "SimulationRadius", want) end)
    local got = nil
    pcall(function() got = tonumber(LocalPlayer.SimulationRadius) end)
    S.simRadReadback = got
    return want
end
S.boostSimRadius()
S.dominateSimRadius = function()
    if not S.SimRadDominate or not S.EngineActive then return end
    if tick() - (S.lastDominateTick or 0) < 0.5 then return end
    S.lastDominateTick = tick()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local skip = false
            if S.WhitelistFriends then pcall(function() skip = LocalPlayer:IsFriendsWith(plr.UserId) end) end
            if not skip then
                pcall(function() plr.MaximumSimulationRadius = 0 end)
                pcall(function() sethiddenproperty(plr, "SimulationRadius", 0) end)
            end
        end
    end
end

    return S.voidFloorY
end

S.checkStuckOwnership = function()
    if not S.EngineActive then return end
    local p = 0
    for _ in pairs(S.pendingCandidateParts) do p = p + 1 end
    S.stuckPending = p
    if #S.verifiedOwnedParts == 0 and p > 0 then
        S.stuckSince = S.stuckSince or tick()
        if (tick() - S.stuckSince) > 4 and (S.escalateStage or 0) < 2 and (tick() - (S.lastEscalateTick or 0)) > 6 then
            S.escalateStage = (S.escalateStage or 0) + 1
            S.lastEscalateTick = tick()
            if S.escalateStage == 1 then
                S.escalatedRadius = math.max((S.OwnershipRadius or 200) * 4, 1500)
            else
                S.escalatedRadius = math.max((S.OwnershipRadius or 200) * 20, 12000)
            end
            S.boostSimRadius()
            Library:Notify(string.format("%d parts pending, ownership not dropping. SimRadius auto-raised to %d (stage %d/2)",
                p, math.floor(S.escalatedRadius), S.escalateStage), 5)
        end
    elseif #S.verifiedOwnedParts > 0 then
        S.escalateStage = 0 S.stuckSince = nil
    end
end

S.stopAnimSlot = function(slot)
    local t = S.animTracks[slot]
    if t and t ~= false then pcall(function() t:Stop(0.15) end) end
    if S.animCurrentSlot == slot then S.animCurrentSlot = nil end
end
S.stopAllAnimTracks = function()
    for slot in pairs(S.animTracks) do
        local tr = S.animTracks[slot]
        if tr and tr ~= false then pcall(function() tr:Stop(0.15) end) end
    end
    S.animCurrentSlot = nil
end
S.resolveAnimId = function(o, rig, key)
    if o and o > 0 then return o end
    local set = S.AnimSets[rig] or S.AnimSets.R6
    return set[key] or set.Idle
end
S.playAnimSlot = function(slot, assetId, force)
    if not S.AnimReplEnabled or not assetId or assetId <= 0 then return end
    if S.animCurrentSlot == slot and not force then return end
    local tr = S.animTracks[slot]
    if tr == false then
        if S.animFailedId[slot] == assetId then return end
        S.animTracks[slot] = nil tr = nil
    end
    if not tr then
        local _, an = S.getHumanoidAndAnimator(LocalPlayer.Character)
        if not an then return end
        local anim = Instance.new("Animation")
        anim.AnimationId = "rbxassetid://" .. tostring(assetId)
        local ok, loaded = pcall(function() return an:LoadAnimation(anim) end)
        anim:Destroy()
        if not (ok and loaded and typeof(loaded) == "Instance") then
            S.animTracks[slot] = false S.animFailedId[slot] = assetId return
        end
        tr = loaded
        tr.Priority = Enum.AnimationPriority.Action
        tr.Looped = true
        S.animTracks[slot] = tr S.animFailedId[slot] = nil
    end
    if S.animCurrentSlot and S.animCurrentSlot ~= slot then
        local prev = S.animTracks[S.animCurrentSlot]
        if prev and prev ~= false then pcall(function() prev:Stop(0.15) end) end
    end
    S.animCurrentSlot = slot
    pcall(function() tr.Looped = true tr:Play(0.15) end)
end
S.updateAnimState = function()
    if not S.AnimReplEnabled then if S.animCurrentSlot then S.stopAllAnimTracks() end return end
    local hum, an = S.getHumanoidAndAnimator(LocalPlayer.Character)
    if not hum or not an or S.isPlayerDead then if S.animCurrentSlot then S.stopAllAnimTracks() end return end
    local rig = (hum.RigType == Enum.HumanoidRigType.R15) and "R15" or "R6"
    if tick() < S.tauntUntil then
        S.playAnimSlot("taunt", S.resolveAnimId(S.AnimTauntId, rig, "Dance"))
    elseif S.PalmActive then
        S.playAnimSlot("palm", S.resolveAnimId(S.AnimPalmId, rig, "Cheer"))
    elseif S.SlashActive then
        S.playAnimSlot("slash", S.resolveAnimId(S.AnimSlashId, rig, "Dance2"))
    elseif S.HollowPurpleActive then
        S.playAnimSlot("purple", S.resolveAnimId(S.AnimPurpleId, rig, "Point"))
    elseif S.NativeFlyActive then
        if hum.MoveDirection.Magnitude > 0.05 then
            S.playAnimSlot("dive", S.resolveAnimId(S.AnimDiveId, rig, "Fall"))
        else
            S.playAnimSlot("hover", S.resolveAnimId(S.AnimHoverId, rig, "Idle"))
        end
    elseif S.ShowcaseActive then
        S.playAnimSlot("showcase", S.resolveAnimId(S.AnimTauntId, rig, "Dance"))
    elseif S.animCurrentSlot then S.stopAllAnimTracks() end
end
S.announceToChat = function(text, bypass)
    if not S.ChatAnnounceEnabled or not text then return end
    local now = tick()
    if not bypass and (now - S.lastAnnounceTick) < math.max(3, S.ChatAnnounceCooldown) then return end
    S.lastAnnounceTick = now
    local sent = false
    pcall(function()
        if tostring(TextChatService.ChatVersion) ~= "Enum.ChatVersion.LegacyChatService" then
            local chs = TextChatService:FindFirstChild("TextChannels")
            local ch = chs and chs:FindFirstChild("RBXGeneral")
            if not ch and chs then
                for _, c in ipairs(chs:GetChildren()) do if c:IsA("TextChannel") then ch = c break end end
            end
            if ch then ch:SendAsync(text) sent = true end
        end
    end)
    if not sent then
        pcall(function()
            StarterGui:SetCore("ChatMakeSystemMessage", {Text=text, Color=Color3.fromRGB(200,120,255)})
        end)
    end
end

end

    S.sensorState[part] = true

end
S.setPartCollide = function(part, want, force)
    if part.CanCollide == want then return end
    if not want or (not S.PartCollideEnabled and not force) then part.CanCollide = false return end
    local root = S.cachedRoot
    if root then
        local d = (part.Position - root.Position).Magnitude
        local spd = 0
        pcall(function() spd = part.AssemblyLinearVelocity.Magnitude end)
        if d < S.FlingGuardRadius or spd > S.FlingCollideMaxSpeed then part.CanCollide = false return end
    end
    part.CanCollide = true
end
S.antiFlingSnapBack = function(root, hum, spd)
    local moved = S.lastSafeCharPos and (root.Position - S.lastSafeCharPos).Magnitude or 0
    if S.AntiFlingEnabled and not S.NativeFlyActive and not S.isPlayerDead
        and moved > S.FlingMaxJump and spd > (S.FlingMaxSpeed * 0.5) then
        local keep = CFrame.new(S.lastSafeCharPos) * (root.CFrame - root.CFrame.Position)
        root.CFrame = keep
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        local st = hum:GetState()
        if st == Enum.HumanoidStateType.Freefall or st == Enum.HumanoidStateType.FallingDown then
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        S.flingEvents = S.flingEvents + 1
        if tick() - (S.lastFlingToast or 0) > 3 then
            S.lastFlingToast = tick()
            Library:Notify("Anti-fling: snapped back! #" .. S.flingEvents, 3)
        end
    elseif not S.isPlayerDead and spd < S.FlingMaxSpeed then
        S.lastSafeCharPos = root.Position
    end
end
S.enforceFastPartNoClip = function(part, root)
    if not root or not part.CanCollide then return end
    local tf = false
    pcall(function() tf = part.AssemblyLinearVelocity.Magnitude > S.FlingCollideMaxSpeed end)
    local tcl = (part.Position - root.Position).Magnitude < S.FlingGuardRadius
    if not tcl and S.headPart and S.headPart.Parent then
        tcl = (part.Position - S.headPart.Position).Magnitude < (S.HeadGuardRadius or 14)
    end
    if tcl or tf then part.CanCollide = false end
end

S.attemptReclaim = function(part)
    local tot = #S.verifiedOwnedParts
    if tot == 0 then return false end
    local budget = math.min(tot, math.max(1, S.AntiStealSample))
    local toucher, bd = nil, math.huge
    for s = 1, budget do
        local c = S.verifiedOwnedParts[((S.reclaimSampleIdx + s - 2) % tot) + 1]
        if c and c.Parent and c ~= part then
            local d = (c.Position - part.Position).Magnitude
            if d < bd then bd = d toucher = c end
        end
    end
    S.reclaimSampleIdx = ((S.reclaimSampleIdx + budget - 1) % tot) + 1
    if not toucher then return false end
    pcall(function()
        local oldT = toucher.CanTouch

            firetouchinterest(toucher, part, 0)
            firetouchinterest(toucher, part, 1)
        end
        toucher.CanTouch = oldT part.CanTouch = false
    end)
    return true
end
S.applySelfImmunity = function(part)
    local char = LocalPlayer.Character
    if not char or S.activeConstraints[part] then return end
    local cons = {}
    for _, mp in ipairs(char:GetDescendants()) do
        if mp:IsA("BasePart") then
            local ncc = Instance.new("NoCollisionConstraint")
            ncc.Part0=part ncc.Part1=mp ncc.Parent=part
            table.insert(cons, ncc)
        end
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root then
        for _, j in ipairs(root:GetJoints()) do
            local other = (j.Part0 == root and j.Part1) or (j.Part1 == root and j.Part0)
            if other and other:IsA("BasePart") and not other:IsDescendantOf(char) then
                local ncc = Instance.new("NoCollisionConstraint")
                ncc.Part0=part ncc.Part1=other ncc.Parent=part
                table.insert(cons, ncc)
            end
        end
    end
    if S.WhitelistFriends then
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= LocalPlayer and LocalPlayer:IsFriendsWith(pl.UserId) and pl.Character then
                for _, fp in ipairs(pl.Character:GetDescendants()) do
                    if fp:IsA("BasePart") then
                        local ncc = Instance.new("NoCollisionConstraint")
                        ncc.Part0=part ncc.Part1=fp ncc.Parent=part
                        table.insert(cons, ncc)
                    end
                end
            end
        end
    end
    S.activeConstraints[part] = cons
end

            end
        end
    end))
end

S.gatherPartsNearby = function(cp, radius, skipBox)
    local res = {}
    local r = math.max(radius or S.OwnershipRadius, 8)
    local ok = pcall(function()
        local pr = OverlapParams.new()
        pr.FilterType = Enum.RaycastFilterType.Exclude
        pr.FilterDescendantsInstances = { LocalPlayer.Character }
        pr.MaxParts = (r > (S.OwnershipRadius or 200)) and 800 or 500
        for _, h in ipairs(workspace:GetPartBoundsInRadius(cp, r, pr)) do table.insert(res, h) end
        if skipBox then return end
        local bh = r + S.TractorBeamDepth
        local bc = cp + Vector3.new(0, (r - S.TractorBeamDepth) * 0.5, 0)
        local bs = Vector3.new(r * 2 + 8, bh, r * 2 + 8)
        for _, h in ipairs(workspace:GetPartBoundsInBox(CFrame.new(bc), bs, pr)) do table.insert(res, h) end
    end)
    if not ok then
        res = {}
        for _, p in ipairs(workspace:GetDescendants()) do if p:IsA("BasePart") then table.insert(res, p) end end
    end
    return res
end
S.considerCandidate = function(p, rp, checked, radius)
    if not (p and p.Parent and p:IsA("BasePart")) then return end
    if checked and checked[p] then return end
    local char = LocalPlayer.Character
    if p.Anchored or not p.AssemblyRootPart or p.AssemblyRootPart.Anchored then return end
    if char and p:IsDescendantOf(char) then return end
    if p.Name == "HumanoidRootPart" then return end
    if S.isOtherPlayerLimb(p) then return end
    if math.max(p.Size.X, math.max(p.Size.Y, p.Size.Z)) > S.MaxPartSize then return end
    if checked then checked[p] = true end
    local pp = p.Position
    local d3 = (pp - rp).Magnitude
    local hd = (Vector3.new(pp.X,0,pp.Z) - Vector3.new(rp.X,0,rp.Z)).Magnitude
    local vd = rp.Y - pp.Y
    local R = math.max(radius or S.OwnershipRadius, 8)
    if not ((d3 <= R) or (hd <= R and vd >= -R and vd <= S.TractorBeamDepth)) then return end
    if S.pendingCandidateParts[p] or S.verifiedSet[p] then return end
    S.pendingCandidateParts[p] = tick()
end
S.scanCandidateParts = function()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or S.isPlayerDead then return end
    local rp = root.Position
    local ck = {}
    for _, p in ipairs(S.gatherPartsNearby(rp, S.OwnershipRadius, false)) do
        S.considerCandidate(p, rp, ck, S.OwnershipRadius)
    end
end
S.farScanParts = function()
    if not S.FarGrabEnabled or (S.perfPressure or 0) >= 3 then return end
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or S.isPlayerDead then return end
    local fr = S.scanRadius()
    if fr <= (S.OwnershipRadius or 200) + 1 then return end
    local rp = root.Position
    local ck = {}
    for _, p in ipairs(S.gatherPartsNearby(rp, fr, true)) do
        S.considerCandidate(p, rp, ck, fr)
    end
end
S.ghostScanSweep = function(rp)
    local b = math.max(50, S.GhostScanBatch)
    if S.perfPressure >= 2 then b = math.max(40, math.floor(b * 0.5)) end
    local proc = 0
    while proc < b do
        if S.ghostIdx > #S.ghostSnapshot then S.ghostIdx = 1 break end
        local o = S.ghostSnapshot[S.ghostIdx]
        S.ghostIdx = S.ghostIdx + 1
        proc = proc + 1
        if o and o.Parent then S.considerCandidate(o, rp, nil, S.scanRadius()) end
    end
end

S.findTargetBySmartQuery = function(q)
    if not q or q == "" then return nil end
    local cl = q:gsub("@", ""):gsub("%s+", ""):lower()
    if cl == "" then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            if not (S.WhitelistFriends and LocalPlayer:IsFriendsWith(p.UserId)) then
                local un = p.Name:lower():gsub("%s+", "")
                local dn = p.DisplayName:lower():gsub("%s+", "")
                if string.find(un, cl, 1, true) or string.find(dn, cl, 1, true) then return p end
            end
        end
    end
    return nil
end
S.performRandomScan = function()
    S.clearBillboards()
    S.CurrentRandomTargets = {}
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root or S.isPlayerDead then return end
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if not (S.WhitelistFriends and LocalPlayer:IsFriendsWith(p.UserId)) then
                local d = (p.Character.HumanoidRootPart.Position - root.Position).Magnitude
                if d <= S.OwnershipRadius then table.insert(list, {Player=p, Dist=d}) end
            end
        end
    end
    table.sort(list, function(a,b) return a.Dist < b.Dist end)
    for i = 1, math.min(#list, S.RandomTargetCount) do
        local tp = list[i].Player
        table.insert(S.CurrentRandomTargets, tp)
        if tp.Character then S.createTargetBillboard(tp.Character, "🎯 RANDOM TARGET LOCK") end
    end
end

    if S.PalmActive or S.isPlayerDead then return end
    local target = S.SelectedTargetPlayer and S.SelectedTargetPlayer.Character
        and S.SelectedTargetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not target then
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local closest, cd = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do

                if not (S.WhitelistFriends and LocalPlayer:IsFriendsWith(p.UserId)) then
                    local d = (p.Character.HumanoidRootPart.Position - root.Position).Magnitude
                    if d < cd then closest = p cd = d end
                end
            end
        end
        if closest then target = closest.Character.HumanoidRootPart end
    end
    if not target then Library:Notify("Select a target first! ❌", 3) return end
    S.PalmTargetRoot = target S.PalmActive = true S.PalmPhase = 1 S.PalmStartTime = tick()
    Library:Notify("Ground smash & uppercut! 🖐️💥", 3)

S.executeTargetAttack = function()
    if S.isPlayerDead then return end
    if S.GojoModeEnabled then S.triggerGojoHollowPurple() return end
    if S.BladeSlashMode then S.triggerBladeSlash() return end
    if S.DivinePalmMode then S.triggerDivinePalmSmash() return end
    local targets = {}
    if S.TargetMode == 1 then
        if S.SelectedTargetPlayer then table.insert(targets, S.SelectedTargetPlayer) end
    elseif S.TargetMode == 2 then
        for p in pairs(S.MultiLockedPlayers) do if p and p.Parent then table.insert(targets, p) end end
    elseif S.TargetMode == 3 then
        if #S.CurrentRandomTargets == 0 then S.performRandomScan() end
        targets = S.CurrentRandomTargets
    elseif S.TargetMode == 4 then
        if S.SuperLockTarget then table.insert(targets, S.SuperLockTarget) end
    end
    if #targets == 0 then return end
    local av = {}
    for _, p in ipairs(S.verifiedOwnedParts) do
        if p and p.Parent and not p.Anchored and not S.activeHomingParts[p] then table.insert(av, p) end
    end
    if #av == 0 then return end
    local pi = 1
    for _, tp in ipairs(targets) do
        if tp and tp.Character then
            local tr = tp.Character:FindFirstChild("HumanoidRootPart")
            if tr then
                for a = 1, S.LaunchAmmoCount do
                    if pi > #av then break end
                    local pt = av[pi]
                    local aa = ((a - 1) - (S.LaunchAmmoCount / 2)) * 0.4
                    local sx = math.sin(aa) * (S.StagingDistance * 0.7)
                    local sy = (a % 2 == 0) and 1.2 or -0.4
                    S.activeHomingParts[pt] = {
                        Phase = S.StagingAttackEnabled and 1 or 2,
                        StartTime = tick(),
                        StagingOffset = Vector3.new(sx, sy, -S.StagingDistance),
                        Target = tr, TargetPlayer = tp, TargetInitialPos = tr.Position
                    }
                    S.enablePartSensors(pt)
                    pt.CanCollide = false
                    pi = pi + 1
                end
            end
        end
    end
end

S.applyNeckGuard = function(char, force)
    if not S.NeckGuardEnabled or not char or not char.Parent then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    pcall(function() hum.RequiresNeck = false end)
    S.headPart = char:FindFirstChild("Head") or S.headPart
    if S.neckChar == char and S.neckTemplate and not force then return end
    S.neckChar = char
    local neck = char:FindFirstChild("Neck", true)
    if neck and neck:IsA("Motor6D") and neck.Part0 and neck.Part1 then
        S.neckTemplate = {Part0=neck.Part0, Part1=neck.Part1, C0=neck.C0, C1=neck.C1}
    elseif not S.neckTemplate then
        S.buildNeckTemplate(char)
    end
end
S.buildNeckTemplate = function(char)
    if not char then return nil end
    local head = char:FindFirstChild("Head")
    local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    if not (head and torso) then return nil end
    S.neckTemplate = {
        Part0 = torso, Part1 = head,
        C0 = CFrame.new(0, torso.Size.Y * 0.5, 0),
        C1 = CFrame.new(0, -head.Size.Y * 0.5, 0)
    }
    return S.neckTemplate
end
S.repairNeck = function(char)
    if not S.NeckGuardEnabled or not char then return false end
    if not S.neckTemplate then S.buildNeckTemplate(char) end
    if not S.neckTemplate then return false end
    local t = S.neckTemplate
    if not (t.Part0 and t.Part0.Parent and t.Part1 and t.Part1.Parent) then return false end
    local h = char:FindFirstChild("Neck", true)
    if h and h:IsA("Motor6D") and h.Part0 and h.Part1 then return false end
    if h then pcall(function() h:Destroy() end) end
    local m = Instance.new("Motor6D")
    m.Name = "Neck" m.Part0 = t.Part0 m.Part1 = t.Part1 m.C0 = t.C0 m.C1 = t.C1
    m.Parent = t.Part0
    return true
end
S.voidGuardTick = function(root)
    if not S.VoidGuardEnabled or not root then return end
    local fy = S.voidGuardY()
    if root.Position.Y < fy then
        local kr = root.CFrame - root.CFrame.Position
        root.CFrame = CFrame.new(Vector3.new(root.Position.X, fy + 6, root.Position.Z)) * kr
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        S.voidSaves = (S.voidSaves or 0) + 1
        if tick() - (S.lastVoidToast or 0) > 5 then
            S.lastVoidToast = tick()
            Library:Notify("Void guard: pulled back (x" .. S.voidSaves .. ")", 3)
        end
    end
end
S.deathGuardTick = function(root, hum)
    if S.isPlayerDead or not hum or not hum.Parent then return end
    local char = hum.Parent
    if S.NeckGuardEnabled then S.applyNeckGuard(char, false) S.repairNeck(char) end
    S.voidGuardTick(root)
    if not S.RescueEnabled then S.lastHealth = hum.Health return end
    local h = hum.Health
    local mh = hum.MaxHealth
    if (S.lastHealth ~= nil) and h < S.lastHealth - 0.5 then
        S.lastHealthDropInfo = "HP " .. math.floor(S.lastHealth) .. "->" .. math.floor(h)
            .. " parts=" .. #S.verifiedOwnedParts .. " y=" .. math.floor(root and root.Position.Y or 0)
    end
    S.lastHealth = h
    if h <= 0 then
        if (S.rescueCount or 0) < (S.RescueMax or 5) and (tick() - (S.lastRescueTick or 0) > 1) then
            S.rescueCount = (S.rescueCount or 0) + 1
            S.lastRescueTick = tick() S.lastHealth = mh
            pcall(function() hum.Health = mh end)
            Library:Notify("Death refused (" .. S.rescueCount .. "/" .. (S.RescueMax or 5) .. ")", 3)
        end
    elseif (S.rescueCount or 0) > 0 and h > mh * 0.6 and (tick() - (S.lastRescueTick or 0) > 6) then
        S.rescueCount = 0
    end
end

S.makeDraggable = function(handle, obj, cond)
    local dragging, ds, sp
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            if cond and not cond() then return end
            dragging = true ds = inp.Position sp = obj.Position
            local ec
            ec = inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then
                    dragging = false if ec then ec:Disconnect() end
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                local d = inp.Position - ds
                obj.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
            end
        end
    end)
end

track(RunService.Heartbeat:Connect(function(dt)
    S.frameCounter = S.frameCounter + 1
    S.renderSkipParity = S.frameCounter % 2
    S.fpsAccum = S.fpsAccum + dt S.fpsFrames = S.fpsFrames + 1
    if S.fpsAccum >= 0.5 then
        S.hudFps = math.floor((S.fpsFrames / S.fpsAccum) + 0.5)
        S.fpsAccum = 0 S.fpsFrames = 0
        if S.AutoQualityEnabled then
            if S.hudFps < 40 then if S.perfPressure < 3 then S.perfPressure = S.perfPressure + 1 end
            elseif S.hudFps > 55 then if S.perfPressure > 0 then S.perfPressure = S.perfPressure - 1 end end
        end
    end
    S.updateAnimState()

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")

    S.cachedRoot = root
    if root and hum then
        local spd = root.AssemblyLinearVelocity.Magnitude
        local prev = S.lastCharSpeed or spd
        if S.AntiFlingEnabled and not S.NativeFlyActive and not S.isPlayerDead then
            if spd > S.FlingMaxSpeed and (spd - prev) > S.FlingSpikeDelta and spd > 0.05 then
                local v = root.AssemblyLinearVelocity
                root.AssemblyLinearVelocity = v.Unit * math.min(spd,
                    ((spd - prev) > S.FlingSpikeDelta) and math.min(S.FlingMaxSpeed * 0.3, 90) or S.FlingMaxSpeed)
                root.AssemblyAngularVelocity = Vector3.zero
                local st = hum:GetState()
                if st == Enum.HumanoidStateType.Freefall or st == Enum.HumanoidStateType.FallingDown
                    or st == Enum.HumanoidStateType.PlatformStanding or st == Enum.HumanoidStateType.Ragdoll then
                    hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                end
                S.flingEvents = S.flingEvents + 1
                if tick() - (S.lastFlingToast or 0) > 3 then
                    S.lastFlingToast = tick()
                    Library:Notify("Fling blocked! #" .. S.flingEvents, 3)
                end
            end
        end
        S.antiFlingSnapBack(root, hum, spd)
        local av = root.AssemblyAngularVelocity
        if S.AntiFlingEnabled and not S.NativeFlyActive and not S.isPlayerDead
            and av.Magnitude > S.FlingMaxAngular and av.Magnitude > 0.05 then
            root.AssemblyAngularVelocity = av.Unit * S.FlingMaxAngular
        end
        S.lastCharSpeed = root.AssemblyLinearVelocity.Magnitude
    end

    if char and root and hum and not S.isPlayerDead then
        if S.NativeFlyActive then
            if not S.FlyAltitudeY then S.FlyAltitudeY = root.Position.Y end
            local cam = workspace.CurrentCamera
            local cc = cam.CFrame
            local ff = Vector3.new(cc.LookVector.X, 0, cc.LookVector.Z)
            if ff.Magnitude < 0.05 then ff = Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z) end
            if ff.Magnitude < 0.05 then ff = Vector3.new(0,0,-1) end
            ff = ff.Unit
            local md = hum.MoveDirection
            local mv = md.Magnitude > 0.05
            local v = 0
            if S.MoveUp or UserInputService:IsKeyDown(Enum.KeyCode.Space) then v = v + 1 end
            if S.MoveDown or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then v = v - 1 end
            S.FlyAltitudeY = S.FlyAltitudeY + (v * S.NativeFlySpeed * dt)
            S.FlyAltitudeY = math.max(S.FlyAltitudeY, S.voidGuardY())
            local nhp = root.Position + (md * S.NativeFlySpeed * dt)
            local tp = Vector3.new(nhp.X, S.FlyAltitudeY, nhp.Z)
            if S.AnimeGodAnimEnabled then
                if mv then
                    local fl = (md.Magnitude > 0) and md.Unit or ff
                    local tilt = S.AnimReplEnabled and -35 or -65
                    root.CFrame = CFrame.lookAt(tp, tp + fl) * CFrame.Angles(math.rad(tilt), 0, 0)
                else
                    local wv = math.sin(tick() * 3) * 0.03
                    S.FlyAltitudeY = S.FlyAltitudeY + wv
                    root.CFrame = CFrame.lookAt(tp, tp + ff)
                end
            else
                root.CFrame = CFrame.new(tp) * root.CFrame.Rotation
            end
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
        if S.WeatherUmbrellaEnabled and S.WeatherRoofPart and S.WeatherRoofPart.Parent and root then
            S.WeatherRoofPart.CFrame = CFrame.new(root.Position + Vector3.new(0, 9, 0))
        end
        if S.AntiNDSDamage and not S.NativeFlyActive then
            local hg = root:FindFirstChildOfClass("BodyGyro") or root:FindFirstChildOfClass("AlignOrientation")
            if not hg then
                root.AssemblyAngularVelocity = Vector3.zero
            elseif root.AssemblyAngularVelocity.Magnitude > 30 then
                root.AssemblyAngularVelocity = root.AssemblyAngularVelocity.Unit * 30
            end
            if not hg and root.AssemblyLinearVelocity.Y < -42 then
                root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, -42, root.AssemblyLinearVelocity.Z)
            end
            if not S.humanoidStatesDisabled[hum] then
                S.humanoidStatesDisabled[hum] = true
                hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
            end
        end
        if S.AntiSitEnabled and hum.Sit then hum.Sit = false hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end

    S.guardAccum = (S.guardAccum or 0) + dt
    if S.guardAccum > 0.25 then
        S.guardAccum = 0
        S.deathGuardTick(root, hum)
    end

    if not S.EngineActive then return end

    if not S.isPlayerDead and (tick() - S.lastScanTime > 1.5) then
        S.lastScanTime = tick()
        S.scanCandidateParts()
        if tick() - (S.lastFarScanTime or 0) > (S.FarScanInterval or 2.5) then
            S.lastFarScanTime = tick()
            S.farScanParts()
        end
    end
    if S.GhostScanEnabled and not S.isPlayerDead and root then
        if (tick() - S.ghostScanTime > math.max(1, S.GhostScanInterval)) or #S.ghostSnapshot == 0 then
            S.ghostScanTime = tick()
            local ok, snap = pcall(function() return workspace:GetDescendants() end)
            if ok and snap and #snap > 0 then S.ghostSnapshot = snap S.ghostIdx = 1 end
        end
        if #S.ghostSnapshot > 0 then S.ghostScanSweep(root.Position) end
    end

    local drones = {}
    if S.HarvesterSwarmEnabled and #S.verifiedOwnedParts > 0 then
        local tot = math.min(#S.verifiedOwnedParts, S.HarvesterDroneCount)
        for d = 1, tot do
            local pick = ((S.harvesterDroneIdx + d - 2) % #S.verifiedOwnedParts) + 1
            table.insert(drones, S.verifiedOwnedParts[pick])
        end
        S.harvesterDroneIdx = ((S.harvesterDroneIdx + tot - 1) % #S.verifiedOwnedParts) + 1
    end

    table.clear(S.pendingKeys)
    for p in pairs(S.pendingCandidateParts) do table.insert(S.pendingKeys, p) end
    local pc = #S.pendingKeys
    local di = 1
    local sb = math.max(1, S.ClaimSweep)
    if S.perfPressure >= 2 then sb = math.max(8, math.floor(sb * 0.5)) end
    local qm = math.clamp(S.PartQuota, 1, 1000)
    local sw = 0
    if pc > 0 then
        for i = 1, pc do
            if sw >= sb then break end
            sw = sw + 1
            local pi = ((S.pendingIdx + i - 2) % pc) + 1
            local p = S.pendingKeys[pi]
            if p and p.Parent and not p.Anchored and p.AssemblyRootPart and not p.AssemblyRootPart.Anchored then
                local dr = nil
                if #drones > 0 then
                    dr = drones[di]
                    if #drones > 1 then
                        local sn = math.min(#drones, 8)
                        local bd = math.huge
                        for s = 1, sn do
                            local c = drones[((di + s - 2) % #drones) + 1]
                            if c and c.Parent then
                                local d = (c.Position - p.Position).Magnitude
                                if d < bd then bd = d dr = c end
                            end
                        end
                    end
                    di = (di % #drones) + 1
                end
                pcall(function()
                    if dr then firetouchinterest(dr, p, 0) firetouchinterest(dr, p, 1) end
                    p.AssemblyLinearVelocity = Vector3.new(0, 0.25, 0)
                end)
                local owned = false
                pcall(function() if isnetworkowner then owned = isnetworkowner(p) end end)
                if owned then
                    S.pendingCandidateParts[p] = nil
                    if #S.verifiedOwnedParts < qm and not S.verifiedSet[p] then
                        S.verifiedSet[p] = true
                        table.insert(S.verifiedOwnedParts, p)
                        p.CanCollide = false
                        S.disablePartSensors(p)
                        if tick() - S.lastClaimSfxTick > 1.2 then
                            S.lastClaimSfxTick = tick()
                            S.playSfx("rbxasset://sounds/electronicpingshort.wav", 0.3, 1.7)
                        end
                    end
                end
            else
                S.pendingCandidateParts[p] = nil
            end
        end
        S.pendingIdx = ((S.pendingIdx + sw - 1) % math.max(pc, 1)) + 1
    else
        S.pendingIdx = 1
    end

    local cb = math.max(5, S.OwnershipCheckPerFrame)
    if S.perfPressure >= 2 then cb = math.max(4, math.floor(cb * 0.5)) end
    local cc = 0
    while cc < cb and #S.verifiedOwnedParts > 0 do
        if S.demoteIdx > #S.verifiedOwnedParts then S.demoteIdx = 1 end
        local p = S.verifiedOwnedParts[S.demoteIdx]
        cc = cc + 1
        if not (p and p.Parent and not p.Anchored and p.AssemblyRootPart and not p.AssemblyRootPart.Anchored) then
            S.clearSelfImmunity(p) S.verifiedSet[p]=nil S.sensorState[p]=nil S.activeHomingParts[p]=nil
            table.remove(S.verifiedOwnedParts, S.demoteIdx)
            if S.demoteIdx > #S.verifiedOwnedParts then S.demoteIdx = 1 end
        else
            local so = true
            pcall(function() if isnetworkowner then so = isnetworkowner(p) end end)
            if so then
                S.demoteIdx = S.demoteIdx + 1
            else
                S.clearSelfImmunity(p) S.verifiedSet[p]=nil S.sensorState[p]=nil S.activeHomingParts[p]=nil
                table.remove(S.verifiedOwnedParts, S.demoteIdx)
                S.enablePartSensors(p)
                S.pendingCandidateParts[p] = tick()
                S.markStolen(p)
                if S.demoteIdx > #S.verifiedOwnedParts then S.demoteIdx = 1 end
            end
        end
    end

    if S.AntiStealEnabled and #S.verifiedOwnedParts > 0 and #S.stolenList > 0 then
        local bg = math.max(1, S.AntiStealSweep)
        local pr = 0
        local gi = S.stolenIdx
        while pr < bg and #S.stolenList > 0 do
            if gi > #S.stolenList then gi = 1 end
            local sp = S.stolenList[gi]
            pr = pr + 1
            local drop = false
            if not (sp and sp.Parent and not sp.Anchored) then
                drop = true
            else
                S.attemptReclaim(sp)
                local on = false
                pcall(function() if isnetworkowner then on = isnetworkowner(sp) end end)
                if on then
                    S.recaptureCount = S.recaptureCount + 1
                    S.pendingCandidateParts[sp] = tick()
                    drop = true
                end
            end
            if drop then
                if sp then S.stolenSet[sp] = nil end
                local li = #S.stolenList
                S.stolenList[gi] = S.stolenList[li] S.stolenList[li] = nil
                if gi > #S.stolenList then gi = 1 end
            else
                gi = gi + 1
            end
        end
        S.stolenIdx = gi
    end

    S.hudUpdateAccum = S.hudUpdateAccum + dt
    if S.hudUpdateAccum > 0.25 then
        S.hudUpdateAccum = 0
        S.checkStuckOwnership()
    end

    for bi = #S.activeBillboards, 1, -1 do
        local bb = S.activeBillboards[bi]
        if not (bb and bb.Parent and bb.Adornee and bb.Adornee.Parent) then
            if bb and bb.Parent then bb:Destroy() end
            table.remove(S.activeBillboards, bi)
        else
            local th = bb.Adornee.Parent:FindFirstChildOfClass("Humanoid")
            if th and th.Health <= 0 then bb:Destroy() table.remove(S.activeBillboards, bi) end
        end
    end

    if S.AutoRandomScan and not S.isPlayerDead and (tick() - S.lastAutoRandomTick > 3.0) then
        S.lastAutoRandomTick = tick()
        S.performRandomScan()
    end
    if S.ShowcaseActive and not S.isPlayerDead then
        if tick() - S.lastShowcaseTick > math.max(3, S.ShowcaseInterval) then
            S.lastShowcaseTick = tick()
            S.CurrentFormation = (S.CurrentFormation % #S.FormationNames) + 1
            Library:Notify("Showcase: " .. S.FormationNames[S.CurrentFormation]
                .. " (" .. S.CurrentFormation .. "/" .. #S.FormationNames .. ")", 3)
            S.playSfx("rbxasset://sounds/electronicpingshort.wav", 0.45, 1.3)
            S.announceToChat("🌀 W7s SHOWCASE: " .. S.FormationNames[S.CurrentFormation] .. "!")
        end
    end

    if S.CenterMode == 2 and (not S.CenterTargetPlayer or not S.CenterTargetPlayer.Parent) then
        S.CenterMode = 1 S.centerCF = nil S.CenterTargetPlayer = nil S.CenterTargetName = ""
        Library:Notify("Center player left — back to you", 3)
    end
    local cpr = nil
    if S.CenterMode == 2 and S.CenterTargetPlayer then cpr = S.getCenterTargetRoot() end
    if root and cpr and S.CenterAutoChase and not S.NativeFlyActive and not S.isPlayerDead then
        local g = (cpr.Position - root.Position).Magnitude
        if g > S.CenterChaseDist then S.centerChasing = true
        elseif g < S.CenterChaseStop then S.centerChasing = false end
        if S.centerChasing and (tick() - S.lastChaseTick > 0.2) then
            S.lastChaseTick = tick()
            local ch = root.Parent and root.Parent:FindFirstChildOfClass("Humanoid")
            if ch then ch:MoveTo(Vector3.new(cpr.Position.X, root.Position.Y, cpr.Position.Z)) end
        end
    else
        S.centerChasing = false
    end

    local total = #S.verifiedOwnedParts
    if total == 0 then return end

    if S.isPlayerDead then
        for _, p in ipairs(S.verifiedOwnedParts) do

                p.CFrame = S.frozenPartPositions[p]
                p.AssemblyLinearVelocity = Vector3.zero
                p.AssemblyAngularVelocity = Vector3.zero
            end

    end

    if S.SuperLockActive and S.SuperLockTarget then

            local tr = tc and tc:FindFirstChild("HumanoidRootPart")

                local cn = 0
                for _, inf in pairs(S.activeHomingParts) do if inf.TargetPlayer == S.SuperLockTarget then cn = cn + 1 end end
                if cn < S.LaunchAmmoCount then S.executeTargetAttack() end
            else
                for p, inf in pairs(S.activeHomingParts) do if inf.TargetPlayer == S.SuperLockTarget then S.activeHomingParts[p] = nil end end
            end
        end

    if S.HollowPurpleActive and root then
        local el = tick() - S.HollowPurpleStart
        if el < 0.8 then
            S.HollowPurpleState = 1
            local cam = workspace.CurrentCamera
            S.HollowPurpleDir = cam.CFrame.LookVector
            S.HollowPurplePos = root.Position + (S.HollowPurpleDir * 5.0) + Vector3.new(0, 1.2, 0)

            S.HollowPurpleState = 2
            S.HollowPurplePos = S.HollowPurplePos + (S.HollowPurpleDir * 250 * dt)

        elseif pe < 0.60 then S.PalmPhase = 3
        elseif pe < 0.70 then S.PalmPhase = 4
        elseif pe < 1.05 then S.PalmPhase = 5

                    local a = S.orbitAngle + (math.floor(pr * 3) * (2 * math.pi / 3)) + (t * 0.2)
                    lo = Vector3.new(math.cos(a) * prr, py, math.sin(a) * prr)

                    local ez = -safeR * 1.6
                    if pr < 0.55 then
                        local k = pr / 0.55
                        local a = S.orbitAngle + k * math.pi * 2

                        local k = (pr - 0.55) / 0.20

                        lo = Vector3.new(math.cos(a) * er * 0.6 + math.cos(sa) * 1.1, eh + math.sin(a) * er * 0.6 + math.sin(sa) * 1.1, ez)
                    else
                        local k = (pr - 0.90) / 0.10
                        lo = Vector3.new(-er * 1.05 + k * er * 2.1, eh + er * 0.62 + math.sin(k * math.pi) * 0.8, ez)

                    local pr = (i - 1) / total
                    local hy = 5.5 + safeR * 0.45
                    local hz = -safeR * 0.15
                    local orr = safeR * 0.42
                    if pr < 0.70 then
                        local k = pr / 0.70
                        local phi = math.acos(1 - 2 * k)
                        local th = math.sqrt(math.max(total, 1) * math.pi) * phi + S.orbitAngle * 3
                        lo = Vector3.new(orr * math.cos(th) * math.sin(phi), hy + orr * math.cos(phi), hz + orr * math.sin(th) * math.sin(phi))
                    else
                        local k = (pr - 0.70) / 0.30
                        local a = -S.orbitAngle * 4 + k * math.pi * 2
                        lo = Vector3.new(math.cos(a) * safeR * 0.62, hy, hz + math.sin(a) * safeR * 0.62)
                    end
                elseif f == 24 then
                    local rc = 9
                    local ri = (i - 1) % rc
                    local wi = math.floor((i - 1) / rc)
                    local prr = math.max(math.ceil(total / rc), 1)
                    local a = S.orbitAngle + (wi / prr) * math.pi * 2
                    local col = ri % 3
                    local row = math.floor(ri / 3)
                    local rr = 1.7
                    lo = Vector3.new((col - 1) * (safeR * 0.5) + math.cos(a) * rr, 2.5 + (row - 1) * (safeR * 0.5) + math.sin(a) * rr, -safeR * 0.5)
                elseif f == 25 then
                    local pr = (i - 1) / total
                    local sw2 = safeR * 0.9
                    local sh = safeR * 1.1 + 5
                    local sz = 0
                    local by = -3
                    if pr < 0.28 then
                        local k = pr / 0.28
                        lo = Vector3.new(-sw2, by + k * sh, sz)
                    elseif pr < 0.56 then
                        local k = (pr - 0.28) / 0.28
                        lo = Vector3.new(sw2, by + k * sh, sz)
                    elseif pr < 0.72 then
                        local k = (pr - 0.56) / 0.16
                        lo = Vector3.new(-sw2 * 1.25 + k * sw2 * 2.5, by + sh, sz)
                    elseif pr < 0.88 then
                        local k = (pr - 0.72) / 0.16
                        lo = Vector3.new(-sw2 * 1.05 + k * sw2 * 2.1, by + sh - safeR * 0.22, sz)
                    else
                        local k = (pr - 0.88) / 0.12
                        local a = S.orbitAngle * 2 + k * math.pi * 2
                        lo = Vector3.new(math.cos(a) * sw2 * 1.3, by + sh - safeR * 0.55, sz + math.sin(a) * sw2 * 0.25)
                    end
                elseif f == 26 then
                    local pr = (i - 1) / total
                    local a = S.orbitAngle * 3 + pr * math.pi * 8
                    local sr = safeR * (0.25 + 0.75 * pr)
                    lo = Vector3.new(math.cos(a) * sr, -3 + pr * safeR * 1.2, math.sin(a) * sr)
                elseif f == 27 then
                    local a = S.orbitAngle * 2.5 + (i / total) * math.pi * 2
                    local sp = (i % 2 == 0) and 1.0 or 0.45
                    local jg = 0.9 + 0.25 * math.sin(tick() * 25 + i)
                    local sr = safeR * sp * jg
                    lo = Vector3.new(math.cos(a) * sr, ((i % 3) - 1) * 1.2, math.sin(a) * sr)
                elseif f == 28 then
                    local pr = (i - 1) / total
                    local sz = -safeR * 0.85
                    local zh = sz + 2.2
                    local za = sz - 1.4
                    local zl = sz + 1.0
                    if pr < 0.34 then
                        local k = pr / 0.34
                        local a = k * math.pi * 2
                        lo = Vector3.new(math.cos(a) * safeR * 0.34, 5.2 + math.sin(a) * safeR * 0.30, sz)
                    elseif pr < 0.50 then
                        local k = (pr - 0.34) / 0.16
                        local a = k * math.pi * 2
                        lo = Vector3.new(math.cos(a) * safeR * 0.17, 11.4 + math.sin(a) * safeR * 0.19, zh)
                    elseif pr < 0.68 then
                        local k = (pr - 0.50) / 0.18
                        lo = Vector3.new(-(safeR * 0.44 + k * safeR * 0.50), 6.2 - k * 6.0, za)
                    elseif pr < 0.86 then
                        local k = (pr - 0.68) / 0.18
                        lo = Vector3.new(safeR * 0.44 + k * safeR * 0.50, 6.2 - k * 6.0, za)
                    else
                        local k = (pr - 0.86) / 0.14
                        local side = (i % 2 == 0) and -1 or 1
                        lo = Vector3.new(side * safeR * 0.20, -2.4 - k * 4.0, zl)
                    end
                end

                local ro = rot3D * lo
                local tPos = (baseC * CFrame.new(ro)).Position
                local cp = part.Position
                local ds = (tPos - cp).Magnitude
                if ds < 0.02 and S.OrbitSpeed <= 0 and not S.NativeFlyActive and charSpeed < 1.5 then
                    if S.sensorState[part] == true then S.disablePartSensors(part) end
                    return
                end
                if ds < safeR * 0.35 and S.lodSkipOrbit(i) then return end
                if S.FarGrabEnabled and ds > (S.FarGrabTetherDist or 110) then
                    S.farDrag(part, cp, tPos, ds)
                    S.farGrabbedParts[part] = true
                else
                    S.farGrabbedParts[part] = nil
                    local pa = eff
                    local vc = S.MaxPartSpeed
                    if ds > safeR * 2 then
                        local ov = math.clamp((ds - safeR * 2) / math.max(safeR, 1), 0, 6)
                        pa = eff * (2 + ov)
                        vc = S.MaxPartSpeed * (1 + ov * 0.5)
                    end
                    local la = math.clamp(pa * dt, 0.05, 1)
                    local np = cp:Lerp(tPos, la)
                    part.CFrame = CFrame.new(np) * part.CFrame.Rotation
                    local vv = (np - cp) / math.max(dt, 0.016)
                    if vv.Magnitude > vc then vv = vv.Unit * vc end
                    part.AssemblyLinearVelocity = vv
                end
                if S.LethalTorque and S.OrbitSpeed > 0 then
                    part.AssemblyAngularVelocity = Vector3.new(200,200,200)
                else
                    part.AssemblyAngularVelocity = Vector3.zero
                end
            end

            if S.CenterOrbitCollide and S.CenterMode == 2 and pivotRoot ~= root then
                if not part.CanCollide then S.setPartCollide(part, true, true) end
            end
            if part.CanCollide then S.enforceFastPartNoClip(part, root) end

            local ws = part.CanCollide or S.activeHomingParts[part] ~= nil or S.strikeSensors
            if ws then
                if S.sensorState[part] ~= true then S.enablePartSensors(part) end
                if part.CanCollide and not S.activeConstraints[part]
                    and S.immunityUsedThisFrame < S.SelfImmunityPerFrame then
                    S.immunityUsedThisFrame = S.immunityUsedThisFrame + 1
                    S.applySelfImmunity(part)
                end
            elseif S.sensorState[part] == true then
                S.disablePartSensors(part)
                if S.activeConstraints[part] then S.clearSelfImmunity(part) end
            end
        end
        iteration()
    end
end))
